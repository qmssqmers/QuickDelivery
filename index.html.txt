            'medium': 0.3,
            'large': 1.0,
            'extra_large': 3.5
        };

        // UI Functions

        };

        // Main Calculation and Save Logic
        window.calculateAndSave = async function(e) {
            e.preventDefault();
            if(!currentUser) {
                showToast("系統未連接數據庫，請稍後再試。");
                return;
            }

            const btn = document.getElementById('submitBtn');
            btn.disabled = true;
            btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> 處理中...';

            const itemName = document.getElementById('itemName').value;
            const origin = document.getElementById('origin').value;
            const weight = parseFloat(document.getElementById('weight').value);
            const volume = document.getElementById('volume').value;
            const isRegret = document.getElementById('isRegret').checked;

            if(!origin || isNaN(weight) || !volume) return;

            // 1. Calculations
            const co2Emission = parseFloat((weight * carbonFactors[origin]).toFixed(2));
            const phoneCharges = Math.round(co2Emission * 121);
            const treeDays = Math.round(co2Emission / 0.057);
            
            let pkgWaste = packagingWaste[volume];
            let totalWaste = pkgWaste;
            if (isRegret) { totalWaste += weight; }
            totalWaste = parseFloat(totalWaste.toFixed(2));

            // 2. Update Local UI Dashboard
            document.getElementById('initialState').classList.add('hidden');
            const dashboard = document.getElementById('resultsDashboard');
            dashboard.classList.remove('hidden');
            
            document.getElementById('co2Result').innerText = co2Emission.toFixed(2);
            document.getElementById('co2Comparison').innerText = `等同於為智能手機充電 ${phoneCharges.toLocaleString()} 次`;
            document.getElementById('wasteResult').innerText = totalWaste.toFixed(2);
            
            if (isRegret) {
                document.getElementById('wasteDetail').innerHTML = `<span class="text-red-500 font-medium">包含 ${pkgWaste}kg 包裝及 ${weight}kg 劣質貨報廢預估</span>`;
            } else {
                document.getElementById('wasteDetail').innerText = `主要為 ${pkgWaste}kg 的包裝物料`;
            }
            document.getElementById('treeResult').innerText = `${treeDays} 天`;

            const insightBox = document.getElementById('insightMessage');
            let insightHTML = `<strong>分析結論：</strong><br>`;
            if (carbonFactors[origin] > 2) {
                insightHTML += `✈️ 來自較遠地區 (碳排系數 ${carbonFactors[origin]})，運輸碳足跡較高。建議非必要可考慮同城或周邊發貨。<br>`;
                insightBox.className = "p-4 rounded-xl border border-orange-200 bg-orange-50 text-orange-800 text-sm leading-relaxed mt-4";
            } else {
                insightHTML += `🚚 發貨地距離較近，有效減少了運輸碳排放。<br>`;
                insightBox.className = "p-4 rounded-xl border border-brand-200 bg-brand-50 text-brand-800 text-sm leading-relaxed mt-4";
            }

            if (isRegret) {
                insightHTML += `⚠️ <strong>警號：</strong> 劣質貨會直接增加堆填區負擔。下次按下付款前，請三思或選擇更有保證的賣家。`;
                insightBox.className = "p-4 rounded-xl border border-red-200 bg-red-50 text-red-800 text-sm leading-relaxed mt-4";
            }

            insightBox.innerHTML = insightHTML;

            // 3. Save to Firestore (Public Data path based on rules)
            try {
                const recordsRef = collection(db, 'artifacts', appId, 'public', 'data', 'shopping_records');
                await addDoc(recordsRef, {
                    itemName: itemName,
                    origin: origin,
                    originLabel: originLabels[origin] || origin,
                    weight: weight,
                    volume: volume,
                    isRegret: isRegret,
                    co2Emission: co2Emission,
                    waste: totalWaste,
                    userId: currentUser.uid,
                    timestamp: Date.now() // Use timestamp for simple sorting in JS
                });
                
                showToast("記錄已成功同步至全網數據庫！", "success");
                // Form reset
                document.getElementById('recordForm').reset();
            } catch (error) {
                console.error("Error saving document: ", error);
                showToast("無法儲存記錄，請重試。");
            } finally {
                btn.disabled = false;
                btn.innerHTML = '<i class="fa-solid fa-cloud-arrow-up"></i> 計算並上傳至全網數據';
            }
        };

        // Firebase Auth & Data Listening
        async function initApp() {
            try {
                // Auth Rule 3: Auth Before Queries
                const token = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;
                if (token) {
                    await signInWithCustomToken(auth, token);
                } else {
                    await signInAnonymously(auth);
                }
                
                currentUser = auth.currentUser;
                console.log("Authenticated as:", currentUser.uid);

                // Start listening to global public records
                const recordsRef = collection(db, 'artifacts', appId, 'public', 'data', 'shopping_records');
                
                onSnapshot(recordsRef, (snapshot) => {
                    if (!currentUser) return;

                    let totalCO2 = 0;
                    let totalWaste = 0;
                    let totalItems = 0;
                    let regretItems = 0;
                    let records = [];

                    snapshot.forEach(doc => {
                        const data = doc.data();
                        records.push(data);
                        totalCO2 += (data.co2Emission || 0);
                        totalWaste += (data.waste || 0);
                        totalItems++;
                        if (data.isRegret) regretItems++;
                    });

                    // Sort records locally by timestamp descending (Auth Rule 2: No Complex Queries)
                    records.sort((a, b) => (b.timestamp || 0) - (a.timestamp || 0));

                    updateGlobalUI(totalCO2, totalWaste, totalItems, regretItems, records);
                }, (error) => {
                    console.error("Error fetching records: ", error);
                });

            } catch (error) {
                console.error("Authentication failed:", error);
                showToast("系統連接失敗");
            }
        }

        function updateGlobalUI(totalCO2, totalWaste, totalItems, regretItems, records) {
            // Update Aggregated Stats
            document.getElementById('g-co2').innerText = totalCO2.toFixed(1);
            document.getElementById('g-waste').innerText = totalWaste.toFixed(1);
            document.getElementById('g-total-items').innerText = totalItems;
            document.getElementById('g-regret-count').innerText = regretItems;
            
            const treeYears = (totalCO2 / 21).toFixed(1); // 21kg per year per tree
            document.getElementById('g-trees').innerText = treeYears;

            const rate = totalItems > 0 ? ((regretItems / totalItems) * 100).toFixed(1) : 0;
            document.getElementById('g-regret-rate').innerText = `${rate}%`;

            // Update Recent Table (Show top 10 recent)
            const tbody = document.getElementById('recentRecordsTable');
            tbody.innerHTML = '';

            if (records.length === 0) {
                tbody.innerHTML = `<tr><td colspan="5" class="px-6 py-8 text-center text-slate-400">目前還沒有任何記錄，成為第一個提交的人吧！</td></tr>`;
                return;
            }

            const recentRecords = records.slice(0, 10);
            recentRecords.forEach(rec => {
                const tr = document.createElement('tr');
                tr.className = "hover:bg-slate-50 transition-colors";
                
                const statusBadge = rec.isRegret 
                    ? `<span class="bg-red-100 text-red-700 px-2 py-1 rounded text-xs font-medium"><i class="fa-solid fa-triangle-exclamation"></i> 劣質/廢物</span>`
                    : `<span class="bg-green-100 text-green-700 px-2 py-1 rounded text-xs font-medium"><i class="fa-solid fa-check"></i> 正常</span>`;

                tr.innerHTML = `
                    <td class="px-6 py-3 border-b border-slate-50 font-medium text-slate-700 truncate max-w-[150px]" title="${rec.itemName}">${rec.itemName || '未知項目'}</td>
                    <td class="px-6 py-3 border-b border-slate-50 text-slate-600">${rec.originLabel || '未知'}</td>
                    <td class="px-6 py-3 border-b border-slate-50 text-right font-mono">${rec.weight} kg</td>
                    <td class="px-6 py-3 border-b border-slate-50 text-right font-mono text-brand-600">+${rec.co2Emission?.toFixed(2)} kg</td>
                    <td class="px-6 py-3 border-b border-slate-50 text-center">${statusBadge}</td>
                `;
                tbody.appendChild(tr);
            });
        }

        // Boot
        window.addEventListener('DOMContentLoaded', initApp);
